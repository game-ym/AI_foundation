```cpp
#include <iostream>
#include <fstream>
#include <string>

int main() {
    // 读取文本文件示例
    std::ifstream textFile("example.txt");
    if (textFile.is_open()) {
        std::string line;
        std::cout << "Reading text file content:\n";
        while (std::getline(textFile, line)) {
            std::cout << line << std::endl;
        }
        textFile.close();
    } else {
        std::cerr << "Unable to open text file.\n";
    }

    // 读取二进制文件示例
    std::ifstream binaryFile("example.bin", std::ios::binary);
    if (binaryFile.is_open()) {
        unsigned char buffer[1024];
        std::size_t bytesRead;
        std::cout << "\nReading binary file content:\n";
        while ((bytesRead = binaryFile.read(buffer, sizeof(buffer)).gcount()) > 0) {
            for (std::size_t i = 0; i < bytesRead; ++i) {
                std::cout << static_cast<int>(buffer[i]) << " ";
            }
            std::cout << std::endl;
        }
        binaryFile.close();
    } else {
        std::cerr << "Unable to open binary file.\n";
    }

    return 0;
}
```
